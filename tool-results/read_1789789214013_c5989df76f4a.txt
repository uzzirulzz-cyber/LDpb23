   162→function NewOrderDialog({ onCreated }: { onCreated: () => void }) {
   163→  const [open, setOpen] = useState(false);
   164→  const [customerName, setCustomerName] = useState("");
   165→  const [customerEmail, setCustomerEmail] = useState("");
   166→  const [paymentMethod, setPaymentMethod] = useState("card");
   167→  const [status, setStatus] = useState("completed");
   168→  const [items, setItems] = useState<{ name: string; qty: string; price: string }[]>([
   169→    { name: "", qty: "1", price: "" },
   170→  ]);
   171→  const [saving, setSaving] = useState(false);
   172→  const { triggerRefresh } = useDashboard();
   173→
   174→  const updateItem = (i: number, field: "name" | "qty" | "price", v: string) =>
   175→    setItems((prev) => prev.map((it, idx) => (idx === i ? { ...it, [field]: v } : it)));
   176→  const removeItem = (i: number) => setItems((prev) => prev.filter((_, idx) => idx !== i));
   177→  const addItem = () => setItems((prev) => [...prev, { name: "", qty: "1", price: "" }]);
   178→
   179→  const total = items.reduce((s, it) => s + (Number(it.qty) || 0) * (Number(it.price) || 0), 0);
   180→
   181→  const reset = () => {
   182→    setCustomerName("");
   183→    setCustomerEmail("");
   184→    setPaymentMethod("card");
   185→    setStatus("completed");
   186→    setItems([{ name: "", qty: "1", price: "" }]);
   187→  };
   188→
   189→  const submit = async () => {
   190→    if (!customerName.trim() || !customerEmail.trim()) {
   191→      toast.error("Customer name and email are required");
   192→      return;
   193→    }
   194→    const cleanItems = items
   195→      .filter((it) => it.name.trim() && Number(it.price) >= 0 && Number(it.qty) > 0)
   196→      .map((it) => ({ name: it.name.trim(), qty: Number(it.qty), price: Number(it.price) }));
   197→    if (cleanItems.length === 0) {
   198→      toast.error("Add at least one valid line item");
   199→      return;
   200→    }
   201→    setSaving(true);
   202→    try {
   203→      const res = await fetch("/api/orders", {
   204→        method: "POST",
   205→        headers: { "Content-Type": "application/json" },
   206→        body: JSON.stringify({ customerName, customerEmail, paymentMethod, status, items: cleanItems }),
   207→      });
   208→      if (!res.ok) throw new Error("Failed to create order");
   209→      toast.success("Order created", { description: `${cleanItems.length} item(s) · $${total.toFixed(2)}` });
   210→      reset();
   211→      setOpen(false);
   212→      triggerRefresh();
   213→      onCreated();
   214→    } catch (e) {
   215→      toast.error("Could not create order", { description: e instanceof Error ? e.message : "" });
   216→    } finally {
   217→      setSaving(false);
   218→    }
   219→  };
   220→
   221→  return (
   222→    <Dialog open={open} onOpenChange={(o) => { setOpen(o); if (!o) reset(); }}>
   223→      <DialogTrigger asChild>
   224→        <Button size="sm" className="gap-1.5">
   225→          <Plus className="h-4 w-4" /> New Order
   226→        </Button>
   227→      </DialogTrigger>
   228→      <DialogContent className="sm:max-w-2xl">
   229→        <DialogHeader>
   230→          <DialogTitle>Create new order</DialogTitle>
   231→        </DialogHeader>
   232→        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   233→          <div className="space-y-1.5">
   234→            <Label htmlFor="o-cust-name">Customer name</Label>
   235→            <Input id="o-cust-name" value={customerName} onChange={(e) => setCustomerName(e.target.value)} placeholder="Jane Doe" />
   236→          </div>
   237→          <div className="space-y-1.5">
   238→            <Label htmlFor="o-cust-email">Customer email</Label>
   239→            <Input id="o-cust-email" type="email" value={customerEmail} onChange={(e) => setCustomerEmail(e.target.value)} placeholder="jane@example.com" />
   240→          </div>
   241→          <div className="space-y-1.5">
   242→            <Label>Payment method</Label>
   243→            <Select value={paymentMethod} onValueChange={setPaymentMethod}>
   244→              <SelectTrigger className="w-full"><SelectValue placeholder="Payment" /></SelectTrigger>
   245→              <SelectContent>
   246→                {PAYMENT_OPTIONS.map((p) => (
   247→                  <SelectItem key={p} value={p} className="capitalize">{PAYMENT_META[p]?.label ?? p}</SelectItem>
   248→                ))}
   249→              </SelectContent>
   250→            </Select>
   251→          </div>
   252→          <div className="space-y-1.5">
   253→            <Label>Status</Label>
   254→            <Select value={status} onValueChange={setStatus}>
   255→              <SelectTrigger className="w-full"><SelectValue placeholder="Status" /></SelectTrigger>
   256→              <SelectContent>
   257→                {STATUS_OPTIONS.filter((s) => s !== "all").map((s) => (
   258→                  <SelectItem key={s} value={s} className="capitalize">{STATUS_META[s]?.label ?? s}</SelectItem>
   259→                ))}
   260→              </SelectContent>
   261→            </Select>
   262→          </div>
   263→        </div>
   264→
   265→        <Separator />
   266→
   267→        <div className="space-y-2">
   268→          <div className="flex items-center justify-between">
   269→            <Label>Line items</Label>
   270→            <Button type="button" variant="outline" size="sm" className="h-7 gap-1.5" onClick={addItem}>
   271→              <Plus className="h-3.5 w-3.5" /> Add item
   272→            </Button>
   273→          </div>
   274→          <div className="space-y-2">
   275→            {items.map((it, i) => (
   276→              <div key={i} className="grid grid-cols-[1fr_70px_90px_32px] items-center gap-2">
   277→                <Input
   278→                  placeholder="Product name"
   279→                  value={it.name}
   280→                  onChange={(e) => updateItem(i, "name", e.target.value)}
   281→                />
   282→                <Input
   283→                  type="number"
   284→                  min={1}
   285→                  placeholder="Qty"
   286→                  value={it.qty}
   287→                  onChange={(e) => updateItem(i, "qty", e.target.value)}
   288→                />
   289→                <Input
   290→                  type="number"
   291→                  min={0}
   292→                  placeholder="Price"
   293→                  value={it.price}
   294→                  onChange={(e) => updateItem(i, "price", e.target.value)}
   295→                />
   296→                <Button
   297→                  type="button"
   298→                  variant="ghost"
   299→                  size="icon"
   300→                  className="h-8 w-8 text-muted-foreground hover:text-rose-500"
   301→                  onClick={() => removeItem(i)}
   302→                  disabled={items.length === 1}
   303→                >
   304→                  <X className="h-4 w-4" />
   305→                </Button>
   306→              </div>
   307→            ))}
   308→          </div>
   309→          <div className="flex items-center justify-end gap-2 pt-1 text-sm">
   310→            <span className="text-muted-foreground">Total:</span>

   316→          <Button variant="outline" onClick={() => { reset(); setOpen(false); }}>Cancel</Button>

   319→          </Button>
   320→        </div>
   321→      </DialogContent>
   322→    </Dialog>
   323→  );

   405→export function OrdersSection() {
   406→  const { data: orders, loading, error } = useDashboardFetch<Order[]>("/api/orders");
   407→  const { displayCurrency, triggerRefresh } = useDashboard();
   408→
   409→  const [search, setSearch] = useState("");
   410→  const [status, setStatus] = useState<string>("all");
   411→  const [selected, setSelected] = useState<Order | null>(null);
   412→  const [detailOpen, setDetailOpen] = useState(false);
   413→
   414→  const filtered = useMemo(() => {
   415→    if (!orders) return [];
   416→    return orders.filter((o) => {
   417→      if (status !== "all" && o.status !== status) return false;
   418→      if (search) {
   419→        const q = search.toLowerCase();
   420→        return (
   421→          o.orderNumber.toLowerCase().includes(q) ||
   422→          o.customerName.toLowerCase().includes(q) ||
   423→          o.customerEmail.toLowerCase().includes(q)
   424→        );
   425→      }
   426→      return true;
   427→    });
   428→  }, [orders, status, search]);
   429→
   430→  const kpis = useMemo(() => {
   431→    if (!orders) return { total: 0, completed: 0, refunded: 0, revenue: 0, aov: 0 };
   432→    const total = orders.length;
   433→    const completed = orders.filter((o) => o.status === "completed").length;
   434→    const refunded = orders.filter((o) => o.status === "refunded").length;
   435→    const revenue = orders
   436→      .filter((o) => o.status === "completed")
   437→      .reduce((s, o) => s + convert(o.total, o.currency, displayCurrency), 0);
   438→    const completedOrders = orders.filter((o) => o.status === "completed");
   439→    const aov = completedOrders.length > 0 ? revenue / completedOrders.length : 0;
   440→    return { total, completed, refunded, revenue, aov };
   441→  }, [orders, displayCurrency]);
   442→
   443→  const statusChart = useMemo(() => {
   444→    if (!orders) return [];
   445→    const map: Record<string, number> = {};
   446→    for (const o of orders) map[o.status] = (map[o.status] ?? 0) + 1;
   447→    return Object.entries(map).map(([k, v]) => ({ name: STATUS_META[k]?.label ?? k, value: v, key: k }));
   448→  }, [orders]);
   449→
   450→  const topCustomers = useMemo(() => {
   451→    if (!orders) return [];
   452→    const map: Record<string, { name: string; count: number; total: number }> = {};
   453→    for (const o of orders) {
   454→      if (!map[o.customerEmail]) map[o.customerEmail] = { name: o.customerName, count: 0, total: 0 };
   455→      map[o.customerEmail].count += 1;
   456→      map[o.customerEmail].total += convert(o.total, o.currency, displayCurrency);
   457→    }
   458→    return Object.values(map).sort((a, b) => b.total - a.total).slice(0, 5);
   459→  }, [orders, displayCurrency]);
   460→
   461→  const openRow = (o: Order) => {
   462→    setSelected(o);
   463→    setDetailOpen(true);
   464→  };
   465→
   466→  const patchStatus = async (id: string, newStatus: string) => {
   467→    try {
   468→      const res = await fetch(`/api/orders/${id}`, {
   469→        method: "PATCH",
   470→        headers: { "Content-Type": "application/json" },
   471→        body: JSON.stringify({ status: newStatus }),
   472→      });
   473→      if (!res.ok) throw new Error("Failed to update");
   474→      toast.success("Order updated", { description: `Status → ${STATUS_META[newStatus]?.label ?? newStatus}` });
   475→      triggerRefresh();
   476→    } catch (e) {
   477→      toast.error("Update failed", { description: e instanceof Error ? e.message : "" });
   478→    }
   479→  };
   480→
   481→  const deleteOrder = async (o: Order) => {
   482→    try {
   483→      const res = await fetch(`/api/orders/${o.id}`, { method: "DELETE" });
   484→      if (!res.ok) throw new Error("Failed to delete");
   485→      toast.success("Order deleted", { description: o.orderNumber });
   486→      triggerRefresh();
   487→    } catch (e) {
   488→      toast.error("Delete failed", { description: e instanceof Error ? e.message : "" });
   489→    }
   490→  };
   491→
   492→  return (
   493→    <div>
   494→      <SectionHeader
   495→        title="Customer Orders"
