   144→// ============================ Count-up hook ============================
   145→function useCountUp(target: number, durationMs = 700): number {
   146→  const [val, setVal] = React.useState(0);
   147→  React.useEffect(() => {
   148→    if (!Number.isFinite(target) || target <= 0) {
   149→      setVal(0);
   150→      return;
   151→    }
   152→    let raf = 0;
   153→    const start = performance.now();
   154→    const from = 0;
   155→    const step = (now: number) => {
   156→      const t = Math.min(1, (now - start) / durationMs);
   157→      const eased = 1 - Math.pow(1 - t, 3);
   158→      setVal(Math.round(from + (target - from) * eased));
   159→      if (t < 1) raf = requestAnimationFrame(step);
   160→    };
   161→    raf = requestAnimationFrame(step);
   162→    return () => cancelAnimationFrame(raf);
   163→  }, [target, durationMs]);
   164→  return val;
   165→}
   166→
   167→function CountUp({
   168→  value,
   169→  format,
   170→}: {
   171→  value: number;
   172→  format?: (n: number) => string;
   173→}) {
   174→  const v = useCountUp(value);
   175→  return <>{format ? format(v) : v.toLocaleString()}</>;
   176→}

   407→    if (!leads || leads.length === 0) return [];
   408→    const m: Record<string, number> = {};

   410→      const d = new Date(l.createdAt);
   411→      if (Number.isNaN(d.getTime())) continue;

   413→      m[key] = (m[key] ?? 0) + 1;
   414→    }

   416→      .map(([month, count]) => ({ month, count }))
   417→      .sort((a, b) => a.month.localeCompare(b.month))
   418→      .slice(-12);

   776→                              : it.status === "error"
   777→                                ? "text-rose-600 dark:text-rose-400"
   778→                                : "text-muted-foreground"
   779→                          )}
   780→                        >
   781→                          {it.status}
   782→                        </span>
   783→                      </span>
   784→                    </div>
   785→                  );
   786→                })}
   787→              </div>
   788→            )}

   790→        </Card>
   791→      </div>
   792→
   793→      {/* Honest helper note */}

   796→        · No mock values shown
   797→      </p>
   798→    </div>
   799→  );
   800→}
