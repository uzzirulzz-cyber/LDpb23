   182→  const [newOpen, setNewOpen] = React.useState(false);
   183→

   185→    if (!orders || orders.length === 0) return null;
   186→    let paid = 0,
   187→      pending = 0,
   188→      refunded = 0,
   189→      revenue = 0;
   190→    for (const o of orders) {
   191→      if (o.status === "refunded" || o.paymentStatus === "refunded") {
   192→        refunded += 1;
   193→        continue;
   194→      }
   195→      if (o.status === "paid" || o.paymentStatus === "paid") {
   196→        paid += 1;
   197→        revenue += o.total;
   198→      } else {
   199→        pending += 1;
   200→      }
   201→    }
   202→    return { total: orders.length, paid, pending, refunded, revenue };
   203→  }, [orders]);

   205→  const filtered = React.useMemo(() => {
   206→    if (!orders) return [];

   208→      if (statusFilter !== "none" && o.status !== statusFilter) return false;

   212→          o.orderNumber,
   213→          o.customer?.name ?? "",
   214→          o.customer?.email ?? "",
   215→        ]
   216→          .join(" ")
   217→          .toLowerCase();

   546→function LicenseKeyChip({ value }: { value: string }) {
   547→  const [copied, setCopied] = React.useState(false);
   548→  const copy = async () => {
   549→    try {
   550→      await navigator.clipboard.writeText(value);
   551→      setCopied(true);
   552→      toast.success("License key copied");
   553→      setTimeout(() => setCopied(false), 1500);
   554→    } catch {
   555→      toast.error("Copy failed");
   556→    }
   557→  };
   558→  return (
   559→    <button
   560→      type="button"
   561→      onClick={copy}
   562→      className="inline-flex items-center gap-1.5 rounded-md border border-primary/30 bg-primary/5 px-2 py-1 font-mono text-[11px] hover:bg-primary/10 transition-colors"
   563→      title="Click to copy"
   564→    >
   565→      <span>{value}</span>
   566→      <Copy className="h-3 w-3 text-muted-foreground" />
   567→      {copied ? (
   568→        <span className="text-emerald-600 dark:text-emerald-400">copied</span>
   569→      ) : null}
   570→    </button>
   571→  );
   572→}

   619→    setProductId("none");
   620→    setQty(1);
   621→    setPaymentMethod("manual");
   622→  };
   623→
   624→  const addItem = () => {
   625→    if (productId === "none") {
   626→      toast.error("Select a product first");
   627→      return;
   628→    }
   629→    if (items.find((i) => i.productId === productId)) {
   630→      toast.error("Product already added — increase quantity instead");
   631→      return;
   632→    }
   633→    setItems([...items, { productId, quantity: Math.max(1, qty) }]);
   634→    setProductId("none");
   635→    setQty(1);
   636→  };
   637→
   638→  const removeItem = (pid: string) => {
   639→    setItems(items.filter((i) => i.productId !== pid));
   640→  };
   641→
   642→  const submit = async () => {
   643→    if (customerMode === "existing" && customerId === "none") {
   644→      toast.error("Select a customer");
   645→      return;
   646→    }
   647→    if (customerMode === "new" && (!newCust.name.trim() || !newCust.email.trim())) {
   648→      toast.error("Customer name and email are required");
   649→      return;
   650→    }
   651→    if (items.length === 0) {
   652→      toast.error("Add at least one line item");
   653→      return;
   654→    }
   655→    setSaving(true);
   656→    try {
   657→      // 1. Resolve customer
   658→      let custId = customerId;
   659→      if (customerMode === "new") {
   660→        const cRes = await fetch("/api/crm/customers", {
   661→          method: "POST",
   662→          headers: { "Content-Type": "application/json" },
   663→          body: JSON.stringify({
   664→            name: newCust.name.trim(),
   665→            email: newCust.email.trim(),
   666→            phone: newCust.phone.trim() || null,
   667→            country: newCust.country.trim() || null,
   668→            city: newCust.city.trim() || null,
   669→          }),
   670→        });
   671→        if (!cRes.ok) throw new Error("Customer creation failed");
   672→        const cJson = await cRes.json();
   673→        custId = cJson.data.id;
   674→      }
   675→
   676→      // 2. Add items to cart (one POST per item; cart API upserts quantity)
   677→      for (const it of items) {
   678→        const r = await fetch("/api/store/cart", {
   679→          method: "POST",
   680→          headers: { "Content-Type": "application/json" },
   681→          body: JSON.stringify({
   682→            customerId: custId,
   683→            productId: it.productId,
   684→            quantity: it.quantity,
   685→          }),
   686→        });
   687→        if (!r.ok) throw new Error("Failed to add item to cart");
   688→      }
   689→
   690→      // 3. Checkout
   691→      const ckRes = await fetch("/api/store/checkout", {
   692→        method: "POST",
   693→        headers: { "Content-Type": "application/json" },
   694→        body: JSON.stringify({
   695→          customerId: custId,
   696→          paymentMethod,
   697→        }),
   698→      });
   699→      if (!ckRes.ok) {
   700→        const j = await ckRes.json().catch(() => ({}));
   701→        throw new Error(j.error || `Checkout failed: ${ckRes.status}`);
   702→      }
   703→      const ckJson = await ckRes.json();
   704→      toast.success(
   705→        `Order ${ckJson.data?.orderNumber ?? "created"} placed`
   706→      );
   707→      reset();
   708→      onOpenChange(false);
   709→      onCreated();
   710→    } catch (e) {
   711→      toast.error(e instanceof Error ? e.message : "Unknown error");
   712→    } finally {
   713→      setSaving(false);
   714→    }
   715→  };
   716→
   717→  return (
   718→    <Dialog open={open} onOpenChange={onOpenChange}>
   719→      <DialogContent className="sm:max-w-2xl">
   720→        <DialogHeader>
   721→          <DialogTitle>New order</DialogTitle>
   722→          <DialogDescription>
   723→            Manual back-office order. Uses storefront checkout flow — creates a
   724→            cart for the customer, then triggers checkout.
   725→          </DialogDescription>
   726→        </DialogHeader>
   727→
   728→        <div className="space-y-4 max-h-[60vh] overflow-y-auto pr-1">
   729→          {/* Customer */}
   730→          <div className="space-y-2">
   731→            <div className="flex items-center gap-2">
   732→              <Button
   733→                type="button"
   734→                size="sm"
   735→                variant={customerMode === "existing" ? "default" : "outline"}
   736→                onClick={() => setCustomerMode("existing")}
   737→              >
   738→                Existing customer
   739→              </Button>
   740→              <Button
   741→                type="button"
   742→                size="sm"
   743→                variant={customerMode === "new" ? "default" : "outline"}
   744→                onClick={() => setCustomerMode("new")}
   745→              >
   746→                New customer
   747→              </Button>
   748→            </div>
   749→            {customerMode === "existing" ? (
   750→              <Select value={customerId} onValueChange={setCustomerId}>
   751→                <SelectTrigger className="w-full">
   752→                  <SelectValue placeholder="Select customer" />
   753→                </SelectTrigger>
   754→                <SelectContent>
   755→                  <SelectItem value="none">Select…</SelectItem>
   756→                  {(customers ?? []).map((c) => (
   757→                    <SelectItem key={c.id} value={c.id}>
   758→                      {c.name} — {c.email}
   759→                    </SelectItem>
   760→                  ))}
   761→                </SelectContent>
   762→              </Select>
   763→            ) : (
   764→              <div className="grid grid-cols-2 gap-3">
   765→                <div className="space-y-1.5">
   766→                  <Label>Name *</Label>
   767→                  <Input
   768→                    value={newCust.name}
   769→                    onChange={(e) =>
   770→                      setNewCust({ ...newCust, name: e.target.value })
   771→                    }
   772→                  />
   773→                </div>
   774→                <div className="space-y-1.5">
   775→                  <Label>Email *</Label>
   776→                  <Input
   777→                    type="email"
   778→                    value={newCust.email}
   779→                    onChange={(e) =>
   780→                      setNewCust({ ...newCust, email: e.target.value })
   781→                    }
   782→                  />
   783→                </div>
   784→                <div className="space-y-1.5">
   785→                  <Label>Phone</Label>
   786→                  <Input
   787→                    value={newCust.phone}
   788→                    onChange={(e) =>
   789→                      setNewCust({ ...newCust, phone: e.target.value })
   790→                    }
   791→                  />
   792→                </div>
   793→                <div className="space-y-1.5">
   794→                  <Label>Country</Label>
   795→                  <Input
   796→                    value={newCust.country}
   797→                    onChange={(e) =>
   798→                      setNewCust({ ...newCust, country: e.target.value })
   799→                    }
   800→                  />
   801→                </div>
   802→              </div>
   803→            )}
   804→          </div>
   805→
   806→          {/* Line items */}
   807→          <div className="space-y-2">
   808→            <Label>Line items</Label>
   809→            <div className="flex items-end gap-2">
   810→              <div className="flex-1">
   811→                <Select value={productId} onValueChange={setProductId}>
   812→                  <SelectTrigger className="w-full">
   813→                    <SelectValue placeholder="Select product" />
   814→                  </SelectTrigger>
   815→                  <SelectContent>
   816→                    <SelectItem value="none">Select…</SelectItem>
   817→                    {products.map((p) => (
   818→                      <SelectItem key={p.id} value={p.id}>
   819→                        {p.name} — {formatMoney(p.price, (p.currency as Currency) ?? "PKR")}{" "}
   820→                        (stock {p.stock})
   821→                      </SelectItem>
   822→                    ))}
   823→                  </SelectContent>
   824→                </Select>
   825→              </div>
   826→              <Input
   827→                type="number"
   828→                min={1}
   829→                value={qty}
   830→                onChange={(e) => setQty(Math.max(1, Number(e.target.value)))}
   831→                className="w-20"
   832→              />
   833→              <Button type="button" size="sm" onClick={addItem}>
   834→                <Plus className="h-4 w-4" />
   835→              </Button>
   836→            </div>
   837→            {items.length === 0 ? (
   838→              <p className="text-xs text-muted-foreground italic">
   839→                No items added yet.
   840→              </p>
   841→            ) : (
   842→              <ul className="space-y-1">
   843→                {items.map((it) => {
   844→                  const p = products.find((x) => x.id === it.productId);
   845→                  return (
   846→                    <li
   847→                      key={it.productId}
   848→                      className="flex items-center justify-between rounded-md border p-2 text-sm"
   849→                    >
   850→                      <div className="min-w-0">
   851→                        <p className="truncate font-medium">{p?.name ?? it.productId}</p>
   852→                        <p className="text-xs text-muted-foreground">
   853→                          qty {it.quantity}
   854→                          {p
   855→                            ? ` · ${formatMoney(p.price * it.quantity, "PKR")}`
   856→                            : ""}
   857→                        </p>
   858→                      </div>
   859→                      <Button
   860→                        type="button"
   861→                        size="icon"
   862→                        variant="ghost"
   863→                        onClick={() => removeItem(it.productId)}
   864→                      >
   865→                        <Trash2 className="h-4 w-4 text-rose-500" />
   866→                      </Button>
   867→                    </li>
   868→                  );
   869→                })}
   870→              </ul>
   871→            )}
   872→          </div>
   873→
   874→          {/* Payment */}
   875→          <div className="space-y-1.5">
   876→            <Label>Payment method</Label>
   877→            <Select value={paymentMethod} onValueChange={setPaymentMethod}>
   878→              <SelectTrigger className="w-full">
   879→                <SelectValue placeholder="Select method" />
   880→              </SelectTrigger>
   881→              <SelectContent>
   882→                {PAYMENT_METHODS.map((m) => (
   883→                  <SelectItem key={m} value={m}>
   884→                    {m}
   885→                  </SelectItem>
   886→                ))}
   887→              </SelectContent>
   888→            </Select>
   889→            <p className="text-xs text-amber-600 dark:text-amber-400">
   890→              Note: no payment gateway is connected — orders will be marked paid
   891→              with paymentId=null and attribution noting &quot;No payment gateway
   892→              connected&quot;.
   893→            </p>
   894→          </div>
   895→        </div>
   896→
   897→        <DialogFooter>
   898→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   899→            Cancel
   900→          </Button>
   901→          <Button onClick={submit} disabled={saving}>
   902→            {saving ? (
   903→              <>
   904→                <Loader2 className="h-4 w-4 mr-1 animate-spin" /> Placing order…
   905→              </>
   906→            ) : (
   907→              "Place order"
   908→            )}
   909→          </Button>
   910→        </DialogFooter>
   911→      </DialogContent>
   912→    </Dialog>
   913→  );
   914→}
